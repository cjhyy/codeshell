45: function object(value: unknown, fields: string[], path: string): Record<string, unknown> {
46:   if (!value || typeof value !== "object" || Array.isArray(value)) throw new UnsupportedShape(path);
47:   const record = value as Record<string, unknown>;
48:   for (const field of Object.keys(record))
49:     if (!fields.includes(field)) throw new UnsupportedShape(`${path}.${field}`, "unknown");
50:   return record;
51: }
52: function requireType(condition: boolean, path: string): asserts condition {
53:   if (!condition) throw new UnsupportedShape(path);
54: }
55: function strings(value: unknown, path: string, limit: number): void {
56:   requireType(Array.isArray(value), path);
57:   requireType(value.length <= limit, path);
58:   value.forEach((item, index) => requireType(typeof item === "string", `${path}.${index}`));
59: }
60: function assertion(value: unknown, path: string): void {
61:   const item = object(value, ["id", "kind", "value", "path"], path);
62:   requireType(typeof item.id === "string", `${path}.id`);
63:   if (item.kind === "contains" || item.kind === "not_contains") {
64:     requireType(!Object.hasOwn(item, "path"), `${path}.path`);
65:     requireType(typeof item.value === "string", `${path}.value`);
66:   } else {
67:     requireType(item.kind === "json_field_equals", `${path}.kind`);
68:     strings(item.path, `${path}.path`, 8);
69:     requireType(
70:       item.value === null ||
71:         typeof item.value === "string" ||
72:         typeof item.value === "boolean" ||
73:         (typeof item.value === "number" && Number.isFinite(item.value)),
74:       `${path}.value`,
75:     );
76:   }
77: }

81:   if (new TextEncoder().encode(text).byteLength > MAX_EDITOR_BYTES)

84:   try {
85:     raw = JSON.parse(text);

87:     return { ok: false, reason: "json", path: "$" };
88:   }
89:   try {
90:     const dataset = object(raw, DATASET_FIELDS, "$");
91:     requireType(dataset.schemaVersion === 1, "$.schemaVersion");
92:     requireType(typeof dataset.title === "string", "$.title");
93:     requireType(typeof dataset.taskFamily === "string", "$.taskFamily");
94:     requireType(Array.isArray(dataset.cases), "$.cases");
95:     requireType(dataset.cases.length <= 200, "$.cases");
96:     dataset.cases.forEach((rawCase, index) => {
97:       const path = `$.cases.${index}`;
98:       const item = object(rawCase, CASE_FIELDS, path);
99:       for (const key of ["id", "sourceGroupId", "input"])
100:         requireType(typeof item[key] === "string", `${path}.${key}`);
101:       requireType(
102:         typeof item.version === "number" && Number.isFinite(item.version),
103:         `${path}.version`,
104:       );
105:       requireType(["real", "synthetic"].includes(String(item.provenance)), `${path}.provenance`);
106:       requireType(
107:         ["target_failure", "regression"].includes(String(item.caseRole)),
108:         `${path}.caseRole`,
109:       );
110:       requireType(["dev", "holdout"].includes(String(item.split)), `${path}.split`);
111:       requireType(

113:         `${path}.readiness`,
114:       );
115:       if (Object.hasOwn(item, "expected"))
116:         requireType(typeof item.expected === "string", `${path}.expected`);
117:       if (Object.hasOwn(item, "evidence")) {
118:         const evidence = object(

127:           `${path}.evidence`,

134:         requireType(Array.isArray(item.rubric), `${path}.rubric`);
135:         requireType(item.rubric.length <= 16, `${path}.rubric`);
136:         item.rubric.forEach((value, rubricIndex) => {

139:           requireType(typeof rubric.id === "string", `${rubricPath}.id`);

145:         });
