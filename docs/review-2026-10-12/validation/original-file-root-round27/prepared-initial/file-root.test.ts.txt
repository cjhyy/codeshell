function mainCallback(channel: string, deps: Record<string, any>) {
  const path = "packages/desktop/src/main/index.ts";
  const text = readFileSync(join(repo, path), "utf8");
  const file = ts.createSourceFile(path, text, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  const matches: any[] = [];
  const visit = (node: any) => {
    if (ts.isCallExpression(node) && node.expression.getText(file) === "ipcMain.handle" && node.arguments[0]?.text === channel) matches.push(node.arguments[1]);
    ts.forEachChild(node, visit);
  };
  visit(file); assert.equal(matches.length, 1, `one real Main callback for ${channel}`);
  const block = matches[0].getText(file);
  extractedCallbacks.push({ path, channel, line: file.getLineAndCharacterOfPosition(matches[0].getStart(file)).line + 1, sha256: createHash("sha256").update(block).digest("hex"), block, capturedDependencies: Object.keys(deps) });
  const output = ts.transpileModule(`const callback = ${block};`, { compilerOptions: { target: ts.ScriptTarget.ES2022, module: ts.ModuleKind.None } }).outputText;
  return new Function(...Object.keys(deps), `${output}\nreturn callback;`)(...Object.values(deps));
}
