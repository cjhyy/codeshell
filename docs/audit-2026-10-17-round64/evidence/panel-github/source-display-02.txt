1: import type { GitPanelAppSourceInput } from "./installer.js";
2: import { PanelAppInstallError } from "./paths.js";
3: 
4: const MAX_SOURCE_PATH = 4_096;
5: 
6: export function normalizeGitPanelAppSource(input: GitPanelAppSourceInput): GitPanelAppSourceInput {
7:   if (!input || input.kind !== "git" || typeof input.url !== "string") {
8:     throw new PanelAppInstallError("GitHub Panel App source is invalid");
9:   }
10:   const raw = input.url.trim();
11:   if (!raw || raw.length > MAX_SOURCE_PATH || raw.includes("\0")) {
12:     throw new PanelAppInstallError("GitHub repository URL is invalid");
13:   }
14:   const urlText = /^[A-Za-z0-9_.-]{1,100}\/[A-Za-z0-9_.-]{1,100}(?:\.git)?\/?$/.test(raw)
15:     ? `https://github.com/${raw}`
16:     : /^github\.com\//i.test(raw)
17:       ? `https://${raw}`
18:       : raw;
19:   let parsed: URL;
20:   try {
21:     parsed = new URL(urlText);
22:   } catch {
23:     throw new PanelAppInstallError("GitHub repository URL is invalid");
24:   }
25:   if (
26:     parsed.protocol !== "https:" ||
27:     parsed.hostname.toLowerCase() !== "github.com" ||
28:     parsed.username ||
29:     parsed.password ||
30:     parsed.port ||
31:     parsed.search ||
32:     parsed.hash
33:   ) {
34:     throw new PanelAppInstallError(
35:       "Panel Apps support public https://github.com repositories only",
36:     );
37:   }
38:   const parts = parsed.pathname.replace(/^\/+|\/+$/g, "").split("/");
39:   if (parts.length < 2 || !parts[0] || !parts[1]) {
40:     throw new PanelAppInstallError("GitHub URL must include owner/repository");
41:   }
42:   const owner = parts[0];
43:   const repo = parts[1].replace(/\.git$/i, "");
44:   if (!/^[A-Za-z0-9_.-]{1,100}$/.test(owner) || !/^[A-Za-z0-9_.-]{1,100}$/.test(repo)) {
45:     throw new PanelAppInstallError("GitHub owner or repository name is invalid");
46:   }
47: 
48:   let urlRef: string | undefined;
49:   let urlSubdir: string | undefined;
50:   if (parts.length > 2) {
51:     if (parts[2] !== "tree" || !parts[3]) {
52:       throw new PanelAppInstallError("GitHub URL must point to a repository or /tree/<ref>/<path>");
53:     }
54:     urlRef = decodeURIComponent(parts[3]);
55:     urlSubdir =
56:       parts.length > 4
57:         ? parts
58:             .slice(4)
59:             .map((part) => decodeURIComponent(part))
60:             .join("/")
61:         : undefined;
62:   }
63:   if ((input.ref && urlRef) || (input.subdir && urlSubdir)) {
64:     throw new PanelAppInstallError(
65:       "GitHub tree URLs cannot be combined with separate ref or subdirectory fields",
66:     );
67:   }
68:   const ref = input.ref?.trim() || urlRef;
69:   const subdir = input.subdir?.trim().replaceAll("\\", "/") || urlSubdir;
70:   if (
71:     ref &&
72:     (!/^[A-Za-z0-9][A-Za-z0-9._/-]{0,254}$/.test(ref) ||
73:       ref.includes("..") ||
74:       ref.includes("//") ||
75:       ref.endsWith("/") ||
76:       ref.endsWith(".lock"))
77:   ) {
78:     throw new PanelAppInstallError("GitHub branch, tag, or commit is invalid");
79:   }
80:   if (subdir) {
81:     const segments = subdir.split("/");
82:     if (
83:       subdir.length > 1_024 ||
84:       subdir.startsWith("/") ||
85:       segments.some((segment) => !segment || segment === "." || segment === "..")
86:     ) {
87:       throw new PanelAppInstallError("GitHub Panel App subdirectory is invalid");
88:     }
89:   }
90:   return {
91:     kind: "git",
92:     url: `https://github.com/${owner}/${repo}.git`,
93:     ...(ref ? { ref } : {}),
94:     ...(subdir ? { subdir } : {}),
95:   };
96: }
