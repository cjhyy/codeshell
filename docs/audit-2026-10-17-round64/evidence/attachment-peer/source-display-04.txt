866: async function manifestFilesInside(records: ManifestRecord[], root: string): Promise<string[]> {
867:   const files = new Set<string>();
868:   for (const record of records) {
869:     if (!record.absPath) continue;
870:     if (await pathExistsInside(record.absPath, root)) files.add(record.absPath);
871:   }
872:   return [...files];
873: }

875: function latestAttachmentState(records: ManifestRecord[]): Map<string, ManifestRecord> {
876:   const latest = new Map<string, ManifestRecord>();
877:   for (const record of records) {
878:     if (!record.id) continue;
879:     if (record.event === "staged") latest.set(record.id, record);
880:     if (record.event === "sent") {
881:       const prev = latest.get(record.id) ?? record;
882:       latest.set(record.id, { ...prev, ...record, status: "sent" });
883:     }
884:     if (record.event === "removedFromDraft" || record.event === "cleanup") latest.delete(record.id);
885:   }
886:   return latest;
