8: /** Atomic FD 3 fragments are reassembled separately for each opaque message id. */
9: export function createNativeModelTraceFrameDecoder(options: { now?: () => number } = {}) {
10:   const partials = new Map<
11:     string,
12:     { part: number; startedAt: number; bytes: number; chunks: Buffer[] }
13:   >();
14:   let partialBytes = 0;
15:   const remove = (id: string) => {
16:     const partial = partials.get(id);
17:     if (partial) partialBytes -= partial.bytes;
18:     partials.delete(id);
19:   };
20:   const invalid = (): never => {
21:     throw new Error("Invalid native model observation fragment");
22:   };
23:   return {
24:     read(line: string): string | undefined {
25:       if (Buffer.byteLength(line) > MAX_FRAME_BYTES) invalid();
26:       const value = JSON.parse(line);
27:       if (value?.type !== "model_trace_chunk") return line;
28:       const now = (options.now ?? Date.now)();
29:       for (const [id, partial] of partials) if (now - partial.startedAt > MAX_AGE_MS) remove(id);
30:       if (
31:         !value ||
32:         typeof value !== "object" ||
33:         Array.isArray(value) ||
34:         Object.keys(value).some(
35:           (key) => !["version", "type", "messageId", "part", "data", "final"].includes(key),
36:         ) ||
37:         value.version !== 1 ||
38:         typeof value.messageId !== "string" ||
39:         !UUID.test(value.messageId) ||
40:         !Number.isSafeInteger(value.part) ||
41:         value.part < 0 ||
42:         value.part > Math.ceil(MAX_FRAME_BYTES / MAX_CHUNK_BYTES) ||
43:         typeof value.final !== "boolean" ||
44:         typeof value.data !== "string" ||
45:         !value.data ||
46:         value.data.length > 256 ||
47:         Buffer.byteLength(line) + 1 > 512 ||
48:         !/^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/.test(value.data)
49:       )
50:         invalid();
51:       const bytes = Buffer.from(value.data, "base64");
52:       if (
53:         !bytes.length ||
54:         bytes.length > MAX_CHUNK_BYTES ||
55:         bytes.toString("base64") !== value.data
56:       )
57:         invalid();
58:       let partial = partials.get(value.messageId);
59:       if (!partial) {
60:         if (value.part !== 0 || partials.size >= MAX_INFLIGHT) invalid();
61:         partial = { part: 0, startedAt: now, bytes: 0, chunks: [] };
62:         partials.set(value.messageId, partial);
63:       }
64:       if (
65:         value.part !== partial.part ||
66:         partial.bytes + bytes.length > MAX_FRAME_BYTES ||
67:         partialBytes + bytes.length > MAX_PARTIAL_BYTES
68:       )
69:         invalid();
70:       partial.part++;
71:       partial.bytes += bytes.length;
72:       partialBytes += bytes.length;
73:       partial.chunks.push(bytes);
74:       if (!value.final) return undefined;
75:       remove(value.messageId);
76:       const decoded = new TextDecoder("utf-8", { fatal: true }).decode(
77:         Buffer.concat(partial.chunks),
78:       );
79:       if (decoded.includes("\n") || decoded.includes("\r")) invalid();
80:       // Validate before returning, so a nested fragment cannot bypass parser limits.
81:       const message = JSON.parse(decoded);
82:       if (message?.type !== "model_trace") invalid();
83:       return decoded;
84:     },
85:     clear() {
86:       partials.clear();
87:       partialBytes = 0;
88:     },
89:   };
90: }

26:       truncated = true;

29:     if (typeof input === "string") {
30:       if (/^data:(image|audio|video|application\/pdf)[^,]*,/i.test(input)) {

35:       if (/^\s*[{[]/.test(scrubbed) && scrubbed.length <= MAX_STRING) {

40:         }
