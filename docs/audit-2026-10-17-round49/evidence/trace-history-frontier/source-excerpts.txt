55:   if (input === undefined) return {};

57:   const result: Record<string, unknown> = {};
58:   for (const [key, value] of Object.entries(query)) {
59:     if (!QUERY_FIELDS.has(key))
60:       throw new ModelTraceRequestError(400, `Unknown model trace query field: ${key}`);
61:     if (value === undefined) continue;
62:     if (key === "callType") {
63:       if (!isModelTraceCallType(value))
64:         throw new ModelTraceRequestError(400, "Invalid model trace callType");
65:     } else if (key === "withoutSession") {
66:       if (typeof value !== "boolean")
67:         throw new ModelTraceRequestError(400, "Invalid model trace withoutSession");
68:     } else if (key === "limit") {
69:       if (!Number.isSafeInteger(value) || (value as number) < 1 || (value as number) > 200)
70:         throw new ModelTraceRequestError(400, "Model trace limit must be between 1 and 200");
71:     } else if (key === "since" || key === "until") {
72:       if (typeof value !== "number" || !Number.isFinite(value) || value < 0)
73:         throw new ModelTraceRequestError(400, `Invalid model trace ${key}`);
74:     } else {
75:       const max = key === "search" || key === "cursor" ? 2048 : 512;
76:       if (typeof value !== "string" || value.length > max || value.includes("\0"))
77:         throw new ModelTraceRequestError(400, `Invalid model trace ${key}`);
78:       if (key === "status" && !["running", "completed", "failed", "cancelled"].includes(value))
79:         throw new ModelTraceRequestError(400, "Invalid model trace status");
80:     }
81:     result[key] = value;
82:   }
