1: /**
2:  * Routes a conversation that has entered a Work Session, ahead of Mimi.
3:  *
4:  * This middleware sits between the deterministic remote commands and
5:  * createMimiPetChat. When the host reports no binding for the conversation it
6:  * calls next() and Mimi handles the message exactly as before; when there is a
7:  * binding it terminates the chain, and the message goes to the Session instead
8:  * of the manager.
9:  *
10:  * Two properties matter more than they look:
11:  *
12:  * - Leaving is decided by the host, never by a model. /mimi has
13:  *   to work when the worker is wedged or the bound Session is unreachable,
14:  *   otherwise a user can be stuck inside a Session with no way out.
15:  * - The reply for an accepted message is not this middleware's job. A Session
16:  *   turn can take minutes, far longer than an inbound HTTP request should be
17:  *   held open, so acceptance is acknowledged immediately and the real answer
18:  *   comes back later through the durable outbox.
19:  */
20:

56: /**

58:  * Without them two anonymous messages on one channel are indistinguishable,
59:  * and a reply could reach the wrong person, so such a message is left to Mimi.
60:  */

62:   return Boolean(message.target?.trim() && message.senderId?.trim());

65: export function createBoundSessionChat(options: BoundSessionChatOptions): ChatMiddleware {
66:   const isDirectMessage =
67:     options.isDirectMessage ?? ((message) => message.isDirectMessage === true);
68:   return async ({ message, reply }, next) => {
69:     if (!message.text.trim() && !message.attachments?.length) return next();
70:     if (!isAddressable(message)) return next();
71: 
72:     // A transport timeout is ambiguous: the Work Session may already have
73:     // accepted this message. Leave retry to the durable inbox instead of also
74:     // handing the same message to Mimi.
75:     const disposition = await options.desktop.routeBoundSessionMessage({
76:       channel: message.channel,
77:       target: message.target,
78:       senderId: message.senderId,
79:       ...(message.messageId ? { messageId: message.messageId } : {}),
80:       text: message.text,
81:       isDirectMessage: isDirectMessage(message),
82:     });
83: 
84:     switch (disposition.kind) {
85:       case "not-bound":
86:         return next();
87:       case "accepted":
88:         // The Session's answer arrives later through the outbox. Only an
89:         // explicit notice (the stale-binding reminder) is sent now.
90:         if (disposition.notice) await reply({ text: disposition.notice });
91:         return;
92:       case "left":
93:       case "status":
94:       case "suspended":
95:         await reply({ text: disposition.text });
96:         return;

48:  * Cross-process single-owner lease shared by the CLI and Desktop lifecycle.
49:  * A dead owner's lock is reclaimed, while a live or unreadable lock fails

51:  */
52: export function acquireGatewayInstanceLock(path: string, owner: string): GatewayInstanceLease {
53:   if (!path || !owner) throw new Error("gateway lock path and owner are required");
54:   if (owner.length > MAX_LOCK_OWNER_CHARS || /[\0\r\n]/u.test(owner)) {
55:     throw new Error("gateway lock owner is invalid");
56:   }
57:   const parent = dirname(path);
58:   mkdirSync(parent, { recursive: true, mode: 0o700 });
59:   const parentInfo = lstatSync(parent);
60:   if (parentInfo.isSymbolicLink() || !parentInfo.isDirectory()) {
61:     throw new Error("gateway lock parent must be a real directory");
62:   }
63: 
64:   const record: GatewayLockRecord = {
65:     version: 1,
66:     pid: process.pid,
67:     owner,
68:     token: randomUUID(),
69:     startedAt: Date.now(),
70:   };
71: 
72:   for (let attempt = 0; attempt < 3; attempt++) {
73:     let fd: number | undefined;
74:     let created = false;
75:     try {
76:       fd = openSync(path, "wx", 0o600);
77:       created = true;
78:       writeFileSync(fd, `${JSON.stringify(record)}\n`, "utf-8");
79:       closeSync(fd);
80:       fd = undefined;
81:       if (process.platform !== "win32") chmodSync(path, 0o600);
82:       let released = false;
83:       return {
84:         path,
85:         owner,
86:         release: () => {
87:           if (released) return;
88:           released = true;
89:           removeOwnedLock(path, record.token);
90:         },
91:       };
92:     } catch (error) {
93:       if (fd !== undefined) closeSync(fd);
94:       if (created) rmSync(path, { force: true });
95:       if ((error as NodeJS.ErrnoException).code !== "EEXIST") throw error;
96:       const existing = readLock(path);
97:       if (!existing || isProcessAlive(existing.pid)) {
98:         throw new GatewayAlreadyRunningError(

100:           existing
101:             ? { pid: existing.pid, owner: existing.owner, startedAt: existing.startedAt }
102:             : undefined,
103:         );
104:       }
105:       // Only reclaim a well-formed lock whose process is demonstrably gone.
106:       removeOwnedLock(path, existing.token);
107:     }
108:   }
109:   throw new GatewayAlreadyRunningError(path);
110: }

112: function readLock(path: string): GatewayLockRecord | undefined {
113:   let fd: number | undefined;
114:   try {
115:     const entry = lstatSync(path);
116:     if (entry.isSymbolicLink() || !entry.isFile() || entry.size > MAX_LOCK_FILE_BYTES) {
117:       return undefined;
118:     }
119:     fd = openSync(path, constants.O_RDONLY | (constants.O_NOFOLLOW ?? 0));
120:     const opened = fstatSync(fd);
121:     if (!opened.isFile() || opened.size > MAX_LOCK_FILE_BYTES) return undefined;
122:     const value = JSON.parse(readFileSync(fd, "utf-8")) as Partial<GatewayLockRecord>;
123:     if (
124:       value.version !== 1 ||
125:       !Number.isSafeInteger(value.pid) ||
126:       (value.pid ?? 0) <= 0 ||
127:       typeof value.owner !== "string" ||
128:       value.owner.length === 0 ||
129:       value.owner.length > MAX_LOCK_OWNER_CHARS ||
130:       /[\0\r\n]/u.test(value.owner) ||
131:       typeof value.token !== "string" ||
132:       !/^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/iu.test(
133:         value.token,
134:       ) ||
135:       typeof value.startedAt !== "number"
136:     ) {
137:       return undefined;
138:     }
139:     return value as GatewayLockRecord;
140:   } catch {
141:     return undefined;
142:   } finally {
143:     if (fd !== undefined) closeSync(fd);
144:   }
145: }
