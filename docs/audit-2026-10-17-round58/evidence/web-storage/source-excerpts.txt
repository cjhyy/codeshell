1: /**
2:  * Thin typed wrapper over localStorage for the mobile remote's persisted
3:  * device identity. Keys keep the historical `cs.*` names so an already-paired
4:  * phone (paired against the old inline UI) keeps working after the rebuild.
5:  */
6: import { translate } from "../i18n/translate.js";
7: import { loadUILanguage } from "./uiLanguage.js";
8: 
9: const K = {
10:   deviceId: "cs.deviceId",
11:   deviceSecret: "cs.deviceSecret",
12:   deviceName: "cs.deviceName",
13: } as const;
14: 
15: /**
16:  * Best-effort localStorage write. On a quota-constrained mobile browser
17:  * (iOS/Android default ~5MB) setItem throws QuotaExceededError; since these
18:  * calls run synchronously inside ws.onopen (the auth handshake), an uncaught
19:  * throw aborts onopen mid-flight → handshake never sent → blank app, no retry.
20:  * Swallow the write failure and let the caller proceed with the in-memory value;
21:  * losing persistence only means a fresh secret/name next session.
22:  */
23: function safeSet(key: string, value: string): void {
24:   try {
25:     localStorage.setItem(key, value);
26:   } catch {
27:     /* quota / disabled storage — best-effort, value still returned to caller */
28:   }
29: }
30: 
31: export const deviceStore = {
32:   getId: (): string => localStorage.getItem(K.deviceId) ?? "",
33:   setId: (v: string): void => safeSet(K.deviceId, v),
34:   clearId: (): void => localStorage.removeItem(K.deviceId),
35: 
36:   /** Get the stable per-browser secret, minting+persisting one on first use.
37:    *  Mirrors the old getSecret() so existing devices keep the same secret. */
38:   getOrCreateSecret: (mint: () => string): string => {
39:     let s = localStorage.getItem(K.deviceSecret);
40:     if (!s) {
41:       s = mint();
42:       safeSet(K.deviceSecret, s);
43:     }
44:     return s;
45:   },
46: 
47:   /** Device display name, defaulting to a platform-derived label on first use. */
48:   getOrCreateName: (): string => {
49:     let n = localStorage.getItem(K.deviceName);
50:     if (!n) {
51:       const platform = (typeof navigator !== "undefined" && navigator.platform) || "Phone";
52:       n = translate(loadUILanguage(), "mobile.storage.browserDevice", { platform });
53:       safeSet(K.deviceName, n);
54:     }
55:     return n;
56:   },
57: } as const;

packages/web/src/lib/deviceCredential.ts
1: /**
2:  * Device credential for the mobile remote.
3:  *
4:  * CONTRACT (do not "improve" by hashing): the main process
5:  * (trusted-device-store.ts) stores and compares the `secretHash` field
6:  * VERBATIM — `d.secretHash === input.secretHash`. There is no SHA-256 on
7:  * either side. The field is named "secretHash" historically but is really an
8:  * opaque shared secret. The phone therefore generates a stable random secret
9:  * per browser and sends it raw as `secretHash` for both pair.complete and
10:  * auth.device. Hashing it here would send main a value it never recorded →
11:  * auth.failed. (See also [[project_beta1_feedback_batch_fixes]]: addDevice
12:  * is get-or-create BY secretHash, so the secret must be stable per browser.)
13:  */
14: export function generateSecret(
15:   randomBytes: (n: number) => Uint8Array = (n) =>
16:     crypto.getRandomValues(new Uint8Array(n)),
17: ): string {
18:   const b = randomBytes(32);
19:   return Array.from(b)
20:     .map((x) => x.toString(16).padStart(2, "0"))
21:     .join("");
22: }

19:   if (!value || typeof value !== "object") return false;
20:   const record = value as EnvironmentDescriptor;
21:   return (
22:     record.version === 1 &&
23:     typeof record.id === "string" &&
24:     UUID.test(record.id) &&
25:     typeof record.name === "string" &&
26:     record.name.trim().length > 0 &&
27:     record.name.length <= 128 &&
28:     !/[\u0000-\u001f\u007f]/.test(record.name) &&
29:     ["desktop", "hub", "project-host"].includes(record.kind) &&
30:     record.entryPath === (record.kind === "desktop" ? "/mobile" : "/")
31:   );
32: }

60: export function readEnvironments(storage: StorageAccess): SavedEnvironment[] {
61:   const raw = storage.getItem(KEY);
62:   if (raw === null) return [];
63:   if (raw.length > 100_000) throw new Error("连接记录过大，请清理当前工作台的连接记录。");
64:   const value: unknown = JSON.parse(raw);
65:   if (!Array.isArray(value) || value.length > LIMIT) throw new Error("连接记录格式无效。");
66:   return value.map(savedEnvironment);
67: }
