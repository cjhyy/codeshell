 204 function shortModel(model: string): string {

 216 function formatElapsed(seconds: number): string {
 217   if (seconds < 60) return `${seconds}s`;
 218   const m = Math.floor(seconds / 60);
 219   const s = seconds % 60;
 220   return `${m}m${s.toString().padStart(2, "0")}s`;
 221 }

 223 function formatNumber(n: number): string {
 224   if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + "M";
 225   if (n >= 1_000) return (n / 1_000).toFixed(1) + "K";
 226   return String(n);
