3:  *
4:  * A "topic segment" is a stretch of Mimi conversation bounded by long idle
5:  * gaps. When a new segment begins we inject a carryover brief (open tasks +

64:   if (input.conclusions.length > 0) {

53: /**
54:  * Locate the closed segment's message window. `closingBoundaryMessageId` is the
55:  * client-message-id the closed segment began at; `nextBoundaryMessageId` is the
56:  * newly-opened segment's first turn (its exclusive end). A missing closing
57:  * boundary means the segment had no keyed first turn — the window starts at 0.
58:  * Indices are into the ordered message list (what Engine.toMessages produces),
59:  * so they map directly onto archiveTurnRange.
60:  */
61: export function locateClosureWindow(
62:   messages: readonly ClosureMessage[],
63:   closingBoundaryMessageId: string | undefined,
64:   nextBoundaryMessageId: string | undefined,
65: ): ClosureWindow | null {
66:   const start = closingBoundaryMessageId
67:     ? indexOfClientMessage(messages, closingBoundaryMessageId)
68:     : 0;
69:   if (start < 0) return null;
70:   const end =
71:     nextBoundaryMessageId !== undefined
72:       ? indexOfClientMessage(messages, nextBoundaryMessageId)
73:       : messages.length;
74:   if (end < 0 || end <= start) return null;
75:   return { messages: messages.slice(start, end), range: { start, end } };
76: }
77: 
78: function indexOfClientMessage(
79:   messages: readonly ClosureMessage[],
80:   clientMessageId: string,
81: ): number {
82:   return messages.findIndex((message) => message.clientMessageId === clientMessageId);
83: }

85: /** Render a window of messages as the extraction prompt input (bounded per turn). */
86: export function buildClosureInput(messages: readonly ClosureMessage[]): string {
87:   return messages
88:     .map((message) => ({ role: message.role, text: message.text.trim() }))
89:     .filter((message) => message.text.length > 0)
90:     .map((message) => {
91:       const speaker = message.role === "user" ? "用户" : "mimi";
92:       const clipped =
93:         message.text.length > MAX_MESSAGE_CHARS
94:           ? `${message.text.slice(0, MAX_MESSAGE_CHARS)}…`
95:           : message.text;
96:       return `${speaker}: ${clipped}`;
97:     })
98:     .join("\n\n");

101: /**
102:  * Parse the aux model's JSON reply, tolerating a leading/trailing code fence or
103:  * prose. Returns null when no usable title+summary can be recovered; memories
104:  * are best-effort and clamped to MAX_MEMORIES.
105:  */
106: export function parseClosureResponse(raw: string): ClosureExtraction | null {
107:   const json = extractFirstJsonObject(raw);
108:   if (!json) return null;
109:   let parsed: unknown;
110:   try {
111:     parsed = JSON.parse(json);
112:   } catch {
113:     return null;
114:   }
115:   if (!parsed || typeof parsed !== "object") return null;
116:   const record = parsed as Record<string, unknown>;
117:   const title = typeof record.title === "string" ? record.title.trim() : "";
118:   const summary = typeof record.summary === "string" ? record.summary.trim() : "";
119:   if (!title || !summary) return null;
120:   const memories = Array.isArray(record.memories)
121:     ? record.memories
122:         .filter((value): value is string => typeof value === "string")
123:         .map((value) => value.trim())
124:         .filter((value) => value.length > 0)
125:         .slice(0, MAX_MEMORIES)
126:     : [];
127:   return { title, summary, memories };
128: }
129: 
130: /** Extract the first balanced `{...}` object from arbitrary model text. */
131: function extractFirstJsonObject(raw: string): string | null {
132:   const text = raw.trim();
133:   const open = text.indexOf("{");
134:   if (open < 0) return null;
135:   let depth = 0;
136:   let inString = false;
137:   let escaped = false;
138:   for (let index = open; index < text.length; index += 1) {
139:     const char = text[index];
140:     if (inString) {
141:       if (escaped) escaped = false;
142:       else if (char === "\\") escaped = true;
143:       else if (char === '"') inString = false;
144:       continue;
145:     }
146:     if (char === '"') inString = true;
147:     else if (char === "{") depth += 1;
148:     else if (char === "}") {
149:       depth -= 1;
150:       if (depth === 0) return text.slice(open, index + 1);
151:     }
152:   }
153:   return null;
154: }

108: /**
109:  * Connects the pure topic-segment primitives (packages/pet) to the real main
110:  * process effects: persistent work memory + the generic range-archival seam.
111:  *
112:  * - onDelegationClosed: a delegated Work Session finished → distill one
113:  *   work-memory entry; if (and only if) a concrete turnRange was supplied,
114:  *   collapse those turns of the Mimi conversation via archiveRange.
115:  * - beginTurn: called before each Mimi chat turn. If the idle gap since the
116:  *   last interaction crosses idleMs, open a fresh segment and return the closed
117:  *   boundary plus a carryover brief; otherwise return undefined. Either way the
118:  *   interaction clock advances.
119:  */
120: export class PetSegmentController {
121:   /**
122:    * Serializes beginTurn so its read-decide-write of lastInteractionAt is
123:    * atomic. Two chat entry points (desktop IPC + IM gateway) hit the same pet
124:    * session; without this, concurrent turns both read the old lastInteractionAt,
125:    * both decide to open a new segment, and both return the same closed boundary
126:    * for archival/distillation.
127:    */
128:   private beginTurnQueue: Promise<unknown> = Promise.resolve();
129:   private manualClearPending = false;
130: 
131:   constructor(private readonly options: PetSegmentControllerOptions) {}
132: 
133:   async onDelegationClosed(closure: PetDelegationClosure): Promise<void> {
134:     const segmentId = this.options.store.activeSegment()?.id ?? "unsegmented";
135:     await this.options.store.append(
136:       buildWorkMemoryEntry({
137:         segmentId,
138:         ...(closure.dedupeKey ? { dedupeKey: closure.dedupeKey } : {}),
139:         objective: closure.objective,
140:         outcome: closure.outcome,
141:         ...(closure.workspace ? { workspace: closure.workspace } : {}),
142:         ...(closure.sessionRef ? { sessionRef: closure.sessionRef } : {}),
143:         at: this.options.now(),
144:       }),
145:     );
146:     if (closure.turnRange) {
147:       await this.options.archiveRange(this.options.petSessionId, closure.turnRange);
148:     }
149:   }
150: 
151:   /**
152:    * Called before each Mimi chat turn. `clientMessageId` is the cross-process
153:    * id of the turn being sent — the only turn identity main can observe (the
154:    * renderer-local Message.id is invisible here). When a long-idle boundary is
155:    * crossed we open a fresh segment keyed to that id so the chat UI can render a
156:    * divider (+ optional brief card) immediately before the turn.
157:    */
158:   async beginTurn(clientMessageId?: string): Promise<PetSegmentTurnStart | undefined> {
159:     // Chain on the queue so overlapping turns run one-at-a-time. A failure in
160:     // one turn must not poison the chain, so swallow the tail's rejection.
161:     const run = this.beginTurnQueue
162:       .catch(() => undefined)
163:       .then(() => this.runBeginTurn(clientMessageId));
164:     this.beginTurnQueue = run.catch(() => undefined);
165:     return run;
166:   }
167: 
168:   /**
169:    * Mark the next real chat turn as a manual topic reset. The actual archive
170:    * must wait for that turn's clientMessageId, which is the durable exclusive
171:    * end anchor used by Core before the model sees any history.
172:    */
173:   async requestContextClear(): Promise<void> {
174:     const run = this.beginTurnQueue
175:       .catch(() => undefined)
176:       .then(async () => {
177:         const now = this.options.now();
178:         if (!this.options.store.activeSegment()) {
179:           await this.options.store.openSegment({
180:             id: `seg-${randomUUID()}`,
181:             startedAt: now,
182:           });
183:         }
184:         await this.options.store.setLastInteractionAt(now);
185:         this.manualClearPending = true;
186:       });
187:     this.beginTurnQueue = run.catch(() => undefined);
188:     await run;
189:   }
190: 
191:   private async runBeginTurn(clientMessageId?: string): Promise<PetSegmentTurnStart | undefined> {
192:     const now = this.options.now();
193:     const lastInteractionAt = this.options.store.lastInteractionAt();
194:     const manualClear = this.manualClearPending;
195:     // The very first interaction has no preceding segment to close, so it only
196:     // establishes the interaction baseline — never a visible boundary. Without
197:     // this guard `shouldStartNewSegment(0, now, idleMs)` is always true and the
198:     // user's first message would render a bare "new topic" divider (with no
199:     // brief, since work memory is still empty). Only a *later* turn that crosses
200:     // the idle gap opens a visible new segment.
201:     const isFirstInteraction = lastInteractionAt === 0;
202:     const openNew =
203:       manualClear ||
204:       (!isFirstInteraction &&
205:         shouldStartNewSegment({ lastInteractionAt, now, idleMs: this.options.idleMs }));
206:     if (!openNew) {
207:       // Establish an invisible base segment on the first observed turn. It has
208:       // no UI boundary, but gives the next long-idle crossing a real segment
209:       // to close and archive from the beginning of the conversation.
210:       if (!this.options.store.activeSegment()) {
211:         await this.options.store.openSegment({
212:           id: `seg-${randomUUID()}`,
213:           startedAt: now,
214:         });
215:       }
216:       await this.options.store.setLastInteractionAt(now);
217:       return undefined;
218:     }
219:     // Capture the segment that is about to close before we append the new one.
220:     const closing = this.options.store.activeSegment();
221:     const closedSegment = closing
222:       ? {
223:           segmentId: closing.id,
224:           ...(closing.boundaryBeforeMessageId
225:             ? { closingBoundaryMessageId: closing.boundaryBeforeMessageId }
226:             : {}),
227:           ...(clientMessageId ? { nextBoundaryMessageId: clientMessageId } : {}),
228:           startedAt: closing.startedAt,
229:           endedAt: now,
230:         }
231:       : undefined;
232:     const brief = manualClear ? "" : this.buildBrief();
233:     const briefText = brief.length > 0 ? brief : undefined;
234:     await this.options.store.openSegment({
235:       id: `seg-${randomUUID()}`,
236:       startedAt: now,
237:       ...(clientMessageId ? { boundaryBeforeMessageId: clientMessageId } : {}),
238:       ...(briefText ? { brief: briefText } : {}),
239:     });
240:     await this.options.store.setLastInteractionAt(now);
241:     if (manualClear) this.manualClearPending = false;
242:     return {
243:       ...(briefText ? { carryoverBrief: briefText } : {}),
244:       ...(closedSegment ? { closedSegment } : {}),
245:       ...(manualClear ? { archiveSummary: PET_MANUAL_CLEAR_ARCHIVE_SUMMARY } : {}),
246:     };
247:   }
248: 
249:   /** Distill a segment only after its exclusive end anchor has been persisted. */
250:   async completeSegmentClosure(closed: PetSegmentClosed): Promise<void> {
251:     await this.options.onSegmentClosed?.(closed);
252:   }
253: 
254:   /** Message-keyed topic-segment boundaries for the Mimi chat UI (oldest → newest). */
255:   segmentBoundaries(): { boundaryBeforeMessageId: string; brief?: string }[] {
256:     return this.options.store.segmentBoundaries();
257:   }
258: 
259:   /** Distill open tasks + recent conclusions from stored work memory. */
260:   private buildBrief(): string {
261:     const entries = this.options.store.entries();
262:     const unfinished = entries
263:       .filter((entry) => entry.outcome === "failed" || entry.outcome === "pending-decided")
264:       .slice(-8)
265:       .map((entry) => ({
266:         objective: entry.objective,
267:         ...(entry.workspace ? { workspace: entry.workspace } : {}),
268:       }));
269:     const conclusions = entries
270:       .filter((entry) => entry.outcome === "completed")
271:       .slice(-8)
272:       .map((entry) => entry.objective);
273:     return buildCarryoverBrief({ unfinished, conclusions });
274:   }
275: }
