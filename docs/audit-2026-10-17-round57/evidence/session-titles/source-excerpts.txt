40: function parseTitleMap(value: unknown, strict = false): TitleMap {
41:   if (!value || typeof value !== "object" || Array.isArray(value)) {
42:     if (strict) throw new Error("session title registry is malformed");
43:     return Object.create(null) as TitleMap;
44:   }
45:   if (strict && Object.keys(value).length > MAX_TITLE_ENTRIES)
46:     throw new Error("session title registry contains too many entries");
47:   const result: TitleMap = Object.create(null) as TitleMap;
48:   for (const [id, title] of Object.entries(value).slice(0, MAX_TITLE_ENTRIES)) {
49:     if (!validSessionId(id) || !validTitle(title)) {
50:       if (strict) throw new Error("session title registry contains malformed entries");
51:       continue;
52:     }
53:     if (strict || title) result[id] = title;
54:   }
55:   return result;
56: }

58: function load(target: string, strict = false): TitleMap {
59:   let descriptor: number | undefined;
60:   try {
61:     const entry = fs.lstatSync(target);
62:     if (entry.isSymbolicLink() || !entry.isFile() || entry.size > MAX_TITLE_FILE_BYTES)
63:       throw new Error("session title registry must be a bounded regular file");
64:     descriptor = fs.openSync(
65:       target,
66:       constants.O_RDONLY | (constants.O_NOFOLLOW ?? 0) | (constants.O_NONBLOCK ?? 0),
67:     );
68:     const opened = fs.fstatSync(descriptor);
69:     if (!opened.isFile() || opened.size > MAX_TITLE_FILE_BYTES)
70:       throw new Error("session title registry must be a bounded regular file");
71:     // A concurrent append must not turn a checked four-megabyte file into an
72:     // unbounded allocation. Read at most the limit plus one sentinel byte.
73:     const buffer = Buffer.alloc(Math.min(opened.size + 1, MAX_TITLE_FILE_BYTES + 1));
74:     let bytes = 0;
75:     while (bytes < buffer.length) {
76:       const count = fs.readSync(descriptor, buffer, bytes, buffer.length - bytes, bytes);
77:       if (!count) break;
78:       bytes += count;
79:     }
80:     if (bytes > opened.size || bytes > MAX_TITLE_FILE_BYTES)
81:       throw new Error("session title registry changed during read");
82:     return parseTitleMap(JSON.parse(buffer.subarray(0, bytes).toString("utf8")), strict);
83:   } catch (error) {
84:     if (strict && (error as NodeJS.ErrnoException).code !== "ENOENT") throw error;
85:     return Object.create(null) as TitleMap;
86:   } finally {
87:     if (descriptor !== undefined) fs.closeSync(descriptor);
88:   }
89: }

91: function save(target: string, map: TitleMap): void {
92:   const temporary = `${target}.${process.pid}.${randomUUID()}.tmp`;
93:   try {
94:     const parent = path.dirname(target);
95:     fs.mkdirSync(parent, { recursive: true, mode: 0o700 });
96:     const parentInfo = fs.lstatSync(parent);
97:     if (parentInfo.isSymbolicLink() || !parentInfo.isDirectory())
98:       throw new Error("session title directory must be a real directory");
99:     try {
100:       const targetInfo = fs.lstatSync(target);
101:       if (targetInfo.isSymbolicLink() || !targetInfo.isFile())
102:         throw new Error("session title target must be a regular file");
103:     } catch (error) {
104:       if ((error as NodeJS.ErrnoException).code !== "ENOENT") throw error;
105:     }
106:     const serialized = `${JSON.stringify(map, null, 2)}\n`;
107:     if (Buffer.byteLength(serialized, "utf8") > MAX_TITLE_FILE_BYTES)
108:       throw new Error("session title registry is too large");
109:     fs.writeFileSync(temporary, serialized, { encoding: "utf8", mode: 0o600, flag: "wx" });
110:     fs.renameSync(temporary, target);
111:   } finally {
112:     try {
113:       fs.rmSync(temporary, { force: true });
114:     } catch {
115:       /* preserve the original write error */
116:     }
117:   }

121: export function createSessionTitlesStore(
122:   file: string,
123:   options: { strictMutations?: boolean } = {},
124: ) {
125:   let mutationQueue: Promise<void> = Promise.resolve();
126:   function serializeMutation(mutation: (target: string) => void): Promise<void> {
127:     const target = file;
128:     const result = mutationQueue.then(() => mutation(target));
129:     mutationQueue = result.catch(() => undefined);
130:     return result;
131:   }

133:   async function listTitles(): Promise<TitleMap> {
134:     await mutationQueue.catch(() => undefined);
135:     return load(file);
136:   }

138:   function setTitle(
139:     id: string,
140:     title: string,
141:     expected: { expectedTitle?: string | null } = {},
142:   ): Promise<void> {
143:     if (!validSessionId(id)) throw new Error("invalid session id");
144:     if (!validTitle(title)) throw new Error("invalid session title");
145:     if (
146:       expected.expectedTitle !== undefined &&
147:       expected.expectedTitle !== null &&
148:       !validTitle(expected.expectedTitle)
149:     )
150:       throw new Error("invalid expected session title");
151:     return serializeMutation((target) => {
152:       const parent = path.dirname(target);
153:       fs.mkdirSync(parent, { recursive: true, mode: 0o700 });
154:       const parentInfo = fs.lstatSync(parent);
155:       if (parentInfo.isSymbolicLink() || !parentInfo.isDirectory()) {
156:         throw new Error("session title directory must be a real directory");
157:       }
158:       // A volume root such as /data is writable while its parent / is not.
159:       // Keep both the lock anchor and its sibling lock inside that volume.
160:       const anchor = path.join(parent, `.${path.basename(target)}.mutex`);
161:       fs.mkdirSync(anchor, { recursive: true, mode: 0o700 });
162:       const anchorInfo = fs.lstatSync(anchor);
163:       if (!anchorInfo.isDirectory() || anchorInfo.isSymbolicLink())
164:         throw new Error("session title lock anchor must be a real directory");
165:       // No await inside this synchronous cross-process lock: a second store
166:       // instance must never block the event loop while this holder waits for I/O.
167:       const release = acquireLockOnPath(anchor);
168:       try {
169:         const map = load(target, options.strictMutations);
170:         if (
171:           expected.expectedTitle !== undefined &&
172:           (map[id] ?? "") !== (expected.expectedTitle ?? "")
173:         )
174:           throw new SessionTitleConflictError();
175:         if (title) {
176:           if (!(id in map) && Object.keys(map).length >= MAX_TITLE_ENTRIES) {
177:             throw new Error("session title registry is full");
178:           }
179:           map[id] = title;
180:         } else delete map[id];
181:         save(target, map);
182:       } finally {
183:         release();
184:       }
185:     });
186:   }
187: 
188:   return { listTitles, setTitle };
189: }
